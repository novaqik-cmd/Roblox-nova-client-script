local FALLBACK_MAX_DISTANCE = 5000
local function getMaxDistance()
    local ok, streaming = pcall(function() return workspace.StreamingEnabled end)
    if ok and streaming then
        local ok2, radius = pcall(function() return workspace.StreamingTargetRadius end)
        if ok2 and type(radius) == "number" and radius > 0 then
            return math.floor(radius)
        end
    end
    return FALLBACK_MAX_DISTANCE
end

local function connect(signal, fn)
    local c = signal:Connect(fn)
    connections[#connections + 1] = c
    return c
end

local function tween(obj, time, props, style, dir)
    local t = TweenService:Create(
        obj,
        TweenInfo.new(time, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

local function create(class, props, children)
    local inst = Instance.new(class)
    local parent
    for k, v in pairs(props or {}) do
        if k == "Parent" then parent = v else inst[k] = v end
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    if parent then inst.Parent = parent end
    return inst
end

local function corner(obj, radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius), Parent = obj })
end

local function stroke(obj, color, thickness, transparency)
    return create("UIStroke", {
        Color = color,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = obj,
    })
end

local function gradient(obj, c1, c2, rotation)
    return create("UIGradient", {
        Color = ColorSequence.new(c1, c2),
        Rotation = rotation or 0,
        Parent = obj,
    })
end

local orderCounter = 0
local function nextOrder()
    orderCounter = orderCounter + 1
    return orderCounter
end

local function rndName()
    local chars = {}
    for i = 1, 12 do chars[i] = string.char(math.random(97, 122)) end
    return table.concat(chars)
end

local function getGuiParent()
    if gethui then
        local ok, ui = pcall(gethui)
        if ok and typeof(ui) == "Instance" then return ui end
    end
    local ok, cg = pcall(function() return game:GetService("CoreGui") end)
    if ok and cg then return cg end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local GuiParent = getGuiParent()

local function mount(inst)
    if syn and syn.protect_gui then pcall(syn.protect_gui, inst) end
    local ok = pcall(function() inst.Parent = GuiParent end)
    if not ok or inst.Parent ~= GuiParent then
        inst.Parent = LocalPlayer:WaitForChild("PlayerGui")
    end
end

local function isTeammate(plr)
    return LocalPlayer.Team ~= nil and plr.Team == LocalPlayer.Team
end

local function getColor(plr)
    if not Settings.TeamColors then return Settings.DefaultColor end
    if isTeammate(plr) then return Settings.TeamColor end
    return Settings.EnemyColor
end

local function removeESP(plr)
    local obj = ESPObjects[plr]
    if obj then
        ESPObjects[plr] = nil
        pcall(function() obj.highlight:Destroy() end)
        pcall(function() obj.gui:Destroy() end)
        pcall(function() obj.boxFrame:Destroy() end)
    end
end

        FillTransparency = 0.55,
        OutlineTransparency = 0,
        Enabled = false,
        Parent = ESPFolder,

        Name = rndName(),

        Size = UDim2.fromOffset(180, 120),
        StudsOffset = Vector3.new(0, 2.2, 0),
        LightInfluence = 0,
        Enabled = false,
        Parent = ESPFolder,

        AnchorPoint = Vector2.new(0.5, 1),
        Position = UDim2.fromScale(0.5, 0.5),
        Size = UDim2.fromScale(1, 0.5),

        TextStrokeTransparency = 0.35,

local function unbind(obj)
    obj.char, obj.hum, obj.root, obj.head = nil, nil, nil, nil
    obj.highlight.Adornee = nil
    obj.gui.Adornee = nil
end

local function bindCharacter(plr, obj)
    local char = plr.Character
    if not char or not char.Parent then
        unbind(obj)
        return
    end
    local root = char:FindFirstChild("HumanoidRootPart")
        or char.PrimaryPart
        or char:FindFirstChildWhichIsA("BasePart")
    if not root then
        unbind(obj)
        return
    end
    obj.char = char
    obj.hum = char:FindFirstChildOfClass("Humanoid")
    obj.root = root
    obj.head = char:FindFirstChild("Head") or root
    obj.highlight.Adornee = char
    obj.gui.Adornee = obj.head
end

local function needsRebind(plr, obj)
    local char = plr.Character
    if char ~= obj.char then return true end
    if not char then return false end
    if not obj.root or not obj.root.Parent then return true end
    if obj.head.Parent == nil then return true end
    if obj.head == obj.root and char:FindFirstChild("Head") then return true end
    if obj.hum then
        if not obj.hum.Parent then return true end
    elseif char:FindFirstChildOfClass("Humanoid") then
        return true
    end
    return false
end

local function scan()
    for plr in pairs(ESPObjects) do
        if plr.Parent ~= Players then removeESP(plr) end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local obj = ESPObjects[plr] or createESP(plr)
            if needsRebind(plr, obj) then bindCharacter(plr, obj) end
        end
    end
end

local function rankChams()
    local list = {}
    for _, obj in pairs(ESPObjects) do
        if obj.active then list[#list + 1] = obj end
    end
    table.sort(list, function(a, b) return a.dist < b.dist end)
    for i, obj in ipairs(list) do obj.rank = i end
end

local function updatePlayer(plr, obj, origin, limit, cam)
    local hum, root = obj.hum, obj.root
    local valid = obj.char ~= nil and root ~= nil and root.Parent ~= nil
    local alive = valid and (hum == nil or (hum.Parent ~= nil and hum.Health > 0))

    local dist = math.huge
    if valid and origin then dist = (root.Position - origin).Magnitude end
    obj.dist = dist

    local active = Settings.Enabled and alive and dist <= limit
    obj.active = active

    local boxVisible = false
    if active and Settings.Boxes and cam then
        local scale = 1
        local okScale, s = pcall(obj.char.GetScale, obj.char)
        if okScale and type(s) == "number" and s > 0 then scale = s end

        local top    = cam:WorldToViewportPoint(root.Position + Vector3.new(0, 2.8 * scale, 0))
        local bottom = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3.2 * scale, 0))
        if top.Z > 0 and bottom.Z > 0 then
            local h = bottom.Y - top.Y
            if h > 2 then
                local w = h * 0.55
                obj.boxFrame.Position = UDim2.fromOffset(math.floor(top.X - w / 2), math.floor(top.Y))
                obj.boxFrame.Size = UDim2.fromOffset(math.floor(w), math.floor(h))
                boxVisible = true
            end
        end
    end
    obj.boxFrame.Visible = boxVisible

    obj.highlight.Enabled = active and Settings.Chams and obj.rank <= MAX_HIGHLIGHTS
    obj.gui.Enabled = active and (Settings.Username or Settings.Distance or Settings.Health)
    if not active then return end

    local color = getColor(plr)
    obj.highlight.FillColor = color
    obj.highlight.OutlineColor = color
    obj.boxStroke.Color = color

    obj.nameLabel.Visible = Settings.Username
    obj.nameLabel.TextColor3 = color

    local parts = {}
    if Settings.Distance and origin then
        parts[#parts + 1] = math.floor(dist) .. "m"
    end

    if hum and Settings.Health then
        local maxHp = math.max(hum.MaxHealth, 1)
        local hp = math.clamp(hum.Health, 0, maxHp)
        local pct = hp / maxHp
        obj.barBG.Visible = true
        obj.barFill.Size = UDim2.fromScale(pct, 1)
        obj.barFill.BackgroundColor3 = Color3.fromHSV(pct * 0.33, 0.75, 1)
        parts[#parts + 1] = math.floor(hp) .. " HP"
    else
        obj.barBG.Visible = false
    end

    obj.infoLabel.Visible = #parts > 0
    obj.infoLabel.Text = table.concat(parts, "  |  ")
end

connect(Players.PlayerRemoving, removeESP)

local scanTimer = SCAN_INTERVAL
connect(RunService.RenderStepped, function(dt)
    local doScan = false
    scanTimer = scanTimer + dt
    if scanTimer >= SCAN_INTERVAL then
        scanTimer = 0
        doScan = true
        pcall(scan)
    end

    local myChar = LocalPlayer.Character
    local myRoot = myChar and (myChar:FindFirstChild("HumanoidRootPart") or myChar.PrimaryPart)
    local cam = workspace.CurrentCamera
    local origin = myRoot and myRoot.Position or (cam and cam.CFrame.Position)

    local limit = (Settings.MaxDistance >= MaxDistanceCap) and math.huge or Settings.MaxDistance

    for plr, obj in pairs(ESPObjects) do
        pcall(updatePlayer, plr, obj, origin, limit, cam)
    end

    if doScan then pcall(rankChams) end
end)

local ICON_LEVELS = 10

local Digits = "0123456789abcdefghijklmnopqrstuvwxyz"
local DigitValue = {}
for i = 1, #Digits do
    DigitValue[string.byte(Digits, i)] = i - 1
end

local hasGroupColor = pcall(function()
    local probe = Instance.new("CanvasGroup")
    probe.GroupColor3 = Color3.new(1, 1, 1)
    probe:Destroy()
end)

local function newIcon(parent, name, color, free)
    local def = IconData[name]
    local size = def.size
    local holder = create("CanvasGroup", {
        Size = UDim2.fromOffset(size, size),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Parent = parent,
    })
    if not free then
        holder.AnchorPoint = Vector2.new(0.5, 0.5)
        holder.Position = UDim2.fromScale(0.5, 0.5)
    end

    local frames = {}
    local data = def.data
    for i = 1, #data, 5 do
        local x = DigitValue[string.byte(data, i)]
        local y = DigitValue[string.byte(data, i + 1)]
        local w = DigitValue[string.byte(data, i + 2)]
        local h = DigitValue[string.byte(data, i + 3)]
        local a = DigitValue[string.byte(data, i + 4)]
        frames[#frames + 1] = create("Frame", {
            Position = UDim2.fromOffset(x, y),
            Size = UDim2.fromOffset(w, h),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 1 - a / ICON_LEVELS,
            BorderSizePixel = 0,
            Parent = holder,
        })
    end

    local icon = { holder = holder }

    function icon.set(c, animate)
        if hasGroupColor then
            if animate then
                tween(holder, 0.2, { GroupColor3 = c })
            else
                holder.GroupColor3 = c
            end
        else
            for _, f in ipairs(frames) do
                if animate then
                    tween(f, 0.2, { BackgroundColor3 = c })
                else
                    f.BackgroundColor3 = c
                end
            end
        end
    end

    icon.set(color)
    return icon
end

local function switchTab(name)
    if currentTab == name then return end
    local prev = currentTab
    currentTab = name

    for tabName, tab in pairs(Tabs) do
        local active = tabName == name
        tab.icon.set(active and Theme.AccentSoft or Theme.SubText, true)
        tween(tab.label, 0.2, { TextColor3 = active and Theme.Text or Theme.SubText })
    end
    tween(TabHighlight, 0.4, { Position = UDim2.fromOffset(Tabs[name].x, 3) })

    if prev then
        tween(Pages[prev], 0.15, { GroupTransparency = 1 })
        task.delay(0.17, function()
            for pageName, p in pairs(Pages) do
                if pageName ~= currentTab then p.Visible = false end
            end
        end)
    end

    local page = Pages[name]
    page.Position = UDim2.fromOffset(0, 14)
    page.GroupTransparency = 1
    page.Visible = true
    tween(page, 0.45, { GroupTransparency = 0, Position = UDim2.fromOffset(0, 0) })
end

local function createSlider(card, opts, min, max, default, step, format, callback)
    opts.height = 92
    opts.controlWidth = 84
    local row, icon, _, tile, tileStroke = createRow(card, opts)
    icon.set(Theme.AccentSoft)
    tile.BackgroundColor3 = Theme.AccentBG
    tileStroke.Color = Theme.AccentLine

    local chip = create("TextLabel", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -18, 0, 15),
        Size = UDim2.fromOffset(70, 30),
        BackgroundColor3 = Theme.Tile,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.AccentSoft,
        Text = "",
        Parent = row,
    })
    corner(chip, 9)
    stroke(chip, Theme.TileStroke, 1)

    local hit = create("Frame", {
        Position = UDim2.fromOffset(18, 62),
        Size = UDim2.new(1, -36, 0, 22),
        BackgroundTransparency = 1,
        Parent = row,
    })
    local track = create("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.fromScale(0, 0.5),
        Size = UDim2.new(1, 0, 0, 6),
        BackgroundColor3 = Theme.Off,
        BorderSizePixel = 0,
        Parent = hit,
    })
    corner(track, 3)
    local fill = create("Frame", {
        Size = UDim2.fromScale(0, 1),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        Parent = track,
    })
    corner(fill, 3)
    gradient(fill, Theme.AccentDark, Theme.AccentViolet, 0)
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromScale(0, 0.5),
        Size = UDim2.fromOffset(16, 16),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        ZIndex = 2,
        Parent = track,
    })
    corner(knob, 8)
    stroke(knob, Theme.Accent, 3, 0.55)

    local range = math.max(max - min, 1)
    local value = math.clamp(default, min, max)

    local function render()
        local pct = (value - min) / range
        fill.Size = UDim2.fromScale(pct, 1)
        knob.Position = UDim2.fromScale(pct, 0.5)
        chip.Text = format(value)
    end
    render()

    local dragging = false
    local function setFromMouse()
        local m = UserInputService:GetMouseLocation()
        local pct = math.clamp((m.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        local v = min + range * pct
        v = math.clamp(math.floor(v / step + 0.5) * step, min, max)
        if pct >= 1 then v = max elseif pct <= 0 then v = min end
        if v ~= value then
            value = v
            render()
            callback(value)
        end
    end

    connect(hit.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            tween(knob, 0.15, { Size = UDim2.fromOffset(20, 20) })
            setFromMouse()
        end
    end)
    connect(UserInputService.InputChanged, function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            setFromMouse()
        end
    end)
    connect(UserInputService.InputEnded, function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
            tween(knob, 0.2, { Size = UDim2.fromOffset(16, 16) })
        end
    end)
end

local Picker = {}
do
    local isOpen, callback = false, nil
    local h, s, v = 0, 1, 1

    local overlay = create("TextButton", {
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Color3.new(0, 0, 0),
        BackgroundTransparency = 1,
        AutoButtonColor = false,
        Text = "",
        Visible = false,
        ZIndex = 50,
        Parent = Main,
    })

        AnchorPoint = Vector2.new(0.5, 0.5),

        Size = UDim2.fromOffset(264, 288),

        BorderSizePixel = 0,
        GroupTransparency = 1,
        Active = true,
        Visible = false,
        Parent = overlay,
    })

    local panelBorder = create("Frame", {
        Position = UDim2.fromOffset(1, 1),
        Size = UDim2.new(1, -2, 1, -2),
        BackgroundTransparency = 1,
        ZIndex = 10,
        Parent = panel,

    corner(panelBorder, 15)
    stroke(panelBorder, Theme.StrokeTop, 1, 0.3)

        Size = UDim2.new(1, -32, 0, 20),

        BorderSizePixel = 0,
        Parent = panel,
    })
    corner(sv, 9)
    local white = create("Frame", {
        Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0, ZIndex = 2, Parent = sv,

        Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.new(0, 0, 0),

    corner(black, 9)
    create("UIGradient", { Rotation = 90, Transparency = NumberSequence.new(1, 0), Parent = black })
    local cursor = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(14, 14),
        BackgroundTransparency = 1,
        ZIndex = 4,
        Parent = sv,
    })
    corner(cursor, 7)
    stroke(cursor, Color3.new(1, 1, 1), 2)

    local hue = create("Frame", {
        Position = UDim2.fromOffset(16, 184),
        Size = UDim2.fromOffset(232, 12),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        Parent = panel,
    })
    corner(hue, 6)
    create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0,     Color3.fromRGB(255, 0, 0)),
            ColorSequenceKeypoint.new(1 / 6, Color3.fromRGB(255, 255, 0)),
            ColorSequenceKeypoint.new(2 / 6, Color3.fromRGB(0, 255, 0)),
            ColorSequenceKeypoint.new(3 / 6, Color3.fromRGB(0, 255, 255)),
            ColorSequenceKeypoint.new(4 / 6, Color3.fromRGB(0, 0, 255)),
            ColorSequenceKeypoint.new(5 / 6, Color3.fromRGB(255, 0, 255)),
            ColorSequenceKeypoint.new(1,     Color3.fromRGB(255, 0, 0)),
        }),
        Parent = hue,
    })
    local hueKnob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(6, 20),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        ZIndex = 2,
        Parent = hue,
    })
    corner(hueKnob, 3)
    stroke(hueKnob, Color3.new(0, 0, 0), 1, 0.4)

    local preview = create("Frame", {
        Position = UDim2.fromOffset(16, 240),
        Size = UDim2.fromOffset(32, 32),
        BorderSizePixel = 0,
        Parent = panel,
    })
    corner(preview, 9)
    stroke(preview, Color3.new(1, 1, 1), 1, 0.8)
    local hex = create("TextBox", {
        Position = UDim2.fromOffset(56, 240),
        Size = UDim2.fromOffset(192, 32),
        BackgroundColor3 = Theme.Tile,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.Text,
        PlaceholderText = "#RRGGBB",
        PlaceholderColor3 = Theme.Muted,
        ClearTextOnFocus = false,
        Text = "",
        Parent = panel,
    })
    corner(hex, 9)
    stroke(hex, Theme.TileStroke, 1)

    local function toHex(c)
        return string.format("#%02X%02X%02X",
            math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
    end

    local function apply()
        local c = Color3.fromHSV(h, s, v)
        sv.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
        cursor.Position = UDim2.fromScale(s, 1 - v)
        hueKnob.Position = UDim2.new(h, 0, 0.5, 0)
        preview.BackgroundColor3 = c
        if not hex:IsFocused() then hex.Text = toHex(c) end
        if callback then callback(c) end
    end

    local presets = {
        Color3.fromRGB(255, 75, 75),
        Color3.fromRGB(255, 150, 60),
        Color3.fromRGB(250, 215, 70),
        Color3.fromRGB(75, 200, 120),
        Color3.fromRGB(60, 200, 230),
        Color3.fromRGB(90, 130, 255),
        Color3.fromRGB(170, 100, 255),
        Color3.fromRGB(255, 255, 255),
    }
    local gap = (232 - #presets * 24) / (#presets - 1)
    for i, c in ipairs(presets) do
        local x = 16 + (i - 1) * (24 + gap)
        local swatch = create("TextButton", {
            Position = UDim2.fromOffset(x, 208),
            Size = UDim2.fromOffset(24, 24),
            BackgroundColor3 = c,
            AutoButtonColor = false,
            Text = "",
            BorderSizePixel = 0,
            Parent = panel,
        })
        corner(swatch, 8)
        stroke(swatch, Color3.new(1, 1, 1), 1, 0.8)
        connect(swatch.MouseEnter, function()
            tween(swatch, 0.12, { Size = UDim2.fromOffset(28, 28), Position = UDim2.fromOffset(x - 2, 206) })
        end)
        connect(swatch.MouseLeave, function()
            tween(swatch, 0.12, { Size = UDim2.fromOffset(24, 24), Position = UDim2.fromOffset(x, 208) })
        end)
        connect(swatch.MouseButton1Click, function()
            h, s, v = c:ToHSV()
            apply()
        end)
    end

    local dragging
    local function updateFromMouse()
        local m = UserInputService:GetMouseLocation()
        if dragging == "sv" then
            s = math.clamp((m.X - sv.AbsolutePosition.X) / sv.AbsoluteSize.X, 0, 1)
            v = 1 - math.clamp((m.Y - sv.AbsolutePosition.Y) / sv.AbsoluteSize.Y, 0, 1)
        elseif dragging == "hue" then
            h = math.clamp((m.X - hue.AbsolutePosition.X) / hue.AbsoluteSize.X, 0, 0.999)
        end
        apply()
    end

    connect(sv.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = "sv"; updateFromMouse()
        end
    end)
    connect(hue.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = "hue"; updateFromMouse()
        end
    end)
    connect(UserInputService.InputChanged, function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            updateFromMouse()
        end
    end)
    connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = nil end
    end)

    connect(hex.FocusLost, function()
        local txt = string.gsub(hex.Text, "#", "")
        if string.match(txt, "^%x%x%x%x%x%x$") then
            local r = tonumber(string.sub(txt, 1, 2), 16)
            local g = tonumber(string.sub(txt, 3, 4), 16)
            local b = tonumber(string.sub(txt, 5, 6), 16)
            h, s, v = Color3.fromRGB(r, g, b):ToHSV()
        end
        apply()
    end)

    function Picker.Open(titleText, color, cb)
        h, s, v = color:ToHSV()
        callback = nil
        titleLabel.Text = titleText
        apply()
        callback = cb
        isOpen = true
        overlay.Visible = true
        panel.Visible = true
        panel.GroupTransparency = 1
        panelScale.Scale = 0.92
        tween(overlay, 0.25, { BackgroundTransparency = 0.45 })
        tween(panel, 0.25, { GroupTransparency = 0 })
        tween(panelScale, 0.4, { Scale = 1 }, Enum.EasingStyle.Back)
    end

    function Picker.Close()
        if not isOpen then return end
        isOpen = false
        callback = nil
        dragging = nil
        tween(overlay, 0.2, { BackgroundTransparency = 1 })
        tween(panel, 0.2, { GroupTransparency = 1 })
        tween(panelScale, 0.2, { Scale = 0.92 })
        task.delay(0.22, function()
            if not isOpen then
                overlay.Visible = false
                panel.Visible = false
            end
        end)
    end

    connect(overlay.MouseButton1Click, Picker.Close)
end

local isListening = false
local function createKeybind(card, opts, default, callback)
    opts.controlWidth = 116
    local row, icon, _, tile, tileStroke = createRow(card, opts)

    local btn = create("TextButton", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -18, 0.5, 0),
        Size = UDim2.fromOffset(108, 30),
        BackgroundColor3 = Theme.Tile,
        AutoButtonColor = false,
        BorderSizePixel = 0,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.Text,
        Text = default.Name,
        Parent = row,
    })
    corner(btn, 9)
    local btnStroke = stroke(btn, Theme.TileStroke, 1)

    local current = default
    local function stopListening()
        btn.Text = current.Name
        tween(btnStroke, 0.2, { Color = Theme.TileStroke })
        tween(btn, 0.2, { TextColor3 = Theme.Text })
        tween(tile, 0.2, { BackgroundColor3 = Theme.Tile })
        tween(tileStroke, 0.2, { Color = Theme.TileStroke })
        icon.set(Theme.SubText, true)
        task.defer(function() isListening = false end)
    end

    connect(btn.MouseButton1Click, function()
        if isListening then return end
        isListening = true
        btn.Text = "press a key..."
        tween(btnStroke, 0.2, { Color = Theme.Accent })
        tween(btn, 0.2, { TextColor3 = Theme.AccentSoft })
        tween(tile, 0.2, { BackgroundColor3 = Theme.AccentBG })
        tween(tileStroke, 0.2, { Color = Theme.AccentLine })
        icon.set(Theme.AccentSoft, true)
    end)

    connect(UserInputService.InputBegan, function(input)
        if not isListening or input.UserInputType ~= Enum.UserInputType.Keyboard then return end
        if input.KeyCode ~= Enum.KeyCode.Escape and input.KeyCode ~= Enum.KeyCode.Unknown then
            current = input.KeyCode
            callback(current)
        end
        stopListening()
    end)
end

local opened = false
setOpen = function(state)
    if opened == state then return end
    opened = state
    if state then
        if not Root.Visible then
            Main.GroupTransparency = 1
            ShadowGroup.GroupTransparency = 1
            Scale.Scale = 0.88
            Root.Visible = true
        end
        tween(Main, 0.35, { GroupTransparency = 0 })
        tween(ShadowGroup, 0.5, { GroupTransparency = 0 })
        tween(Scale, 0.55, { Scale = 1 }, Enum.EasingStyle.Back)
    else
        Picker.Close()
        tween(Main, 0.22, { GroupTransparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
        tween(ShadowGroup, 0.22, { GroupTransparency = 1 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
        tween(Scale, 0.28, { Scale = 0.9 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
        task.delay(0.3, function()
            if not opened then Root.Visible = false end
        end)
    end
end

connect(UserInputService.InputBegan, function(input)
    if unloaded or isListening then return end
    if UserInputService:GetFocusedTextBox() then return end
    if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == Settings.Keybind then
        setOpen(not opened)
    end
end)

do
    local dragging, dragStart, startPos
    connect(DragHandle.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = UserInputService:GetMouseLocation()
            startPos = Root.Position
        end
    end)
    connect(UserInputService.InputChanged, function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = UserInputService:GetMouseLocation() - dragStart
            Root.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
    connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
end

Unload = function()
    if unloaded then return end
    setOpen(false)
    task.delay(0.35, function()
        unloaded = true
        for _, c in ipairs(connections) do
            pcall(function() c:Disconnect() end)
        end
        for plr in pairs(ESPObjects) do removeESP(plr) end
        pcall(function() ESPFolder:Destroy() end)
        pcall(function() BoxGui:Destroy() end)
        pcall(function() ScreenGui:Destroy() end)
    end)
    env.NovaClientUnload = nil
end
env.NovaClientUnload = Unload
